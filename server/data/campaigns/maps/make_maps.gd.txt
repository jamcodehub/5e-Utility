func _init() -> void:
	var made := 0
	for spec in SPECS:
		var cfg := MapConfig.new()
		cfg.style = spec[3]
		cfg.width = SIZES[spec[4]][0]
		cfg.height = SIZES[spec[4]][1]
		cfg.theme = spec[5]
		cfg.map_seed = spec[6]
		var skin := ""
		for k in spec[8]:
			if k == "reskin":
				skin = spec[8][k]
			else:
				cfg.set(k, spec[8][k])
		var m = MapGenerator.generate(cfg)
		_reskin(m, skin)
		var d: Dictionary = m.to_site_dict(spec[2])
		var out := {"title": spec[2], "slug": spec[1], "theme": d["theme"], "darkMode": spec[7], "content": d["content"], "doorStates": d["doorStates"]}
		var dir := "%s/%s" % [OUT, spec[0]]
		DirAccess.make_dir_recursive_absolute(dir)
		var f := FileAccess.open("%s/%s.json" % [dir, spec[1]], FileAccess.WRITE)
		f.store_string(JSON.stringify(out))
		f.close()
		made += 1
	print("made %d maps" % made)
	quit()

## Puts a generated outdoor map into a different climate: salt flats (no grass or trees) or winter (snow instead of grass and sand).
func _reskin(m, mode: String) -> void:
	if mode == "":
		return
	for y in m.height:
		for x in m.width:
			var g: String = m.get_ground(x, y)
			var o: String = m.get_object(x, y)
			var h := absi(((x * 73856093) ^ (y * 19349663)) % 100)
			if mode == "salt":
				if g == "grass" or g == "mossy-cobblestone":
					m.set_ground(x, y, "sand" if h < 85 else "rocky-dirt")
				if o == "tree" or o == "bush" or o == "yellow-flowers":
					m.set_object(x, y, "rock-small" if h < 25 else ("rock-medium" if h < 33 else ""))
			elif mode == "winter":
				if g == "grass" or g == "sand":
					m.set_ground(x, y, "snow" if h < 92 else "rocky-dirt")
				elif g == "mossy-cobblestone":
					m.set_ground(x, y, "floor")   # paving, not moss, in the cold
				if o == "yellow-flowers":
					m.set_object(x, y, "")
